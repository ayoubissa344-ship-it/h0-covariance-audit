cmake_minimum_required(VERSION 3.16)
project(h0audit VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
add_library(h0audit INTERFACE)
target_include_directories(h0audit INTERFACE cpp/include)
target_link_libraries(h0audit INTERFACE Threads::Threads)
add_executable(h0audit_cli cpp/src/h0audit_cli.cpp)
set_target_properties(h0audit_cli PROPERTIES OUTPUT_NAME h0audit)
target_link_libraries(h0audit_cli PRIVATE h0audit)
target_compile_options(h0audit_cli PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(h0audit_tests cpp/tests/test_main.cpp)
target_link_libraries(h0audit_tests PRIVATE h0audit)
target_compile_options(h0audit_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME cpp_unit COMMAND h0audit_tests)
install(TARGETS h0audit_cli RUNTIME DESTINATION bin)
install(DIRECTORY cpp/include/ DESTINATION include)
